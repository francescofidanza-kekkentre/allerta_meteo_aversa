// Caricamento 100% dinamico dell'allerta dal file alert.json async function loadAlert() { try { const response = await fetch('./alert.json?v=' + new Date().getTime()); if (response.ok) { const data = await response.json(); const banner = document.getElementById('alertBanner'); const detailsEl = document.getElementById('alertDetailsText'); // Imposta la classe colore (status-ARANCIONE, status-GIALLA, etc.) banner.className = 'alert-banner status-' + data.status; // Titoli e Badge dinamici in base allo stato let icon = '✅'; let title = 'Nessuna allerta meteo critica in vigore'; let badgeText = 'Verde'; if (data.status === 'ARANCIONE') { icon = '⚠️'; title = 'Allerta meteo arancione per rischio idrogeologico e temporali'; badgeText = 'Moderata'; } else if (data.status === 'GIALLA') { icon = '⚠️'; title = 'Allerta meteo gialla per temporali'; badgeText = 'Ordinaria'; } else if (data.status === 'ROSSA') { icon = '🔴'; title = 'ALLERTA METEO ROSSA - CRITICITÀ ELEVATA'; badgeText = 'Elevata'; } // Inserimento DINAMICO dei dati presi da alert.json banner.innerHTML = `
${icon} ${title}
${badgeText}
${data.message}
`; // Mostra la descrizione estesa solo se c'è un'allerta attiva if (data.status !== 'VERDE') { detailsEl.style.display = 'block'; detailsEl.textContent = `Avviso ufficiale Protezione Civile Campania (Zona 2 - Agro Aversano): ${data.message}. Verificare eventuali ordinanze comunali per il territorio di Aversa.`; } else { detailsEl.style.display = 'none'; } document.getElementById('updateTime').textContent = 'Ultima verifica ufficiale: ' + data.updated_at; } } catch (e) { console.warn("Lettura alert.json fallita, uso stato predefinito.", e); } }
